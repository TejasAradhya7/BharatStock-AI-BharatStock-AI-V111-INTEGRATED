from __future__ import annotations
import re
import pandas as pd
from .engine import indicators

def validate_symbol(symbol: str) -> str:
    symbol = symbol.strip().upper()
    if not symbol or len(symbol) > 24 or not re.fullmatch(r"[A-Z0-9.^=_-]+", symbol):
        raise ValueError("Enter a valid ticker symbol, e.g. RELIANCE.NS or ^NSEI.")
    return symbol

def fetch_history(symbol: str, period: str = "1y") -> pd.DataFrame:
    symbol = validate_symbol(symbol)
    # Yahoo Finance tickers for NSE commonly use .NS; NIFTY index is ^NSEI.
    # Import the provider lazily so API health/docs and sentiment remain available
    # even when optional market-data dependencies have not been installed yet.
    try:
        import yfinance as yf
    except ImportError as exc:
        raise RuntimeError("Market-data dependency missing. Install services/api/requirements.txt.") from exc
    df = yf.Ticker(symbol).history(period=period, auto_adjust=True)
    if df is None or df.empty:
        raise LookupError(f"No market history returned for {symbol}. Check the ticker or provider availability.")
    df = df.rename(columns={c: c.title() for c in df.columns})
    cols = ["Open", "High", "Low", "Close", "Volume"]
    missing = set(cols) - set(df.columns)
    if missing:
        raise ValueError(f"Provider response missing columns: {sorted(missing)}")
    df = df[cols].dropna(subset=["Open", "High", "Low", "Close"])
    df.index = pd.to_datetime(df.index)
    return indicators(df)
